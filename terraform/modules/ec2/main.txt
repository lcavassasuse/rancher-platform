data "aws_ami" "opensuse" {
  most_recent = true
  owners      = ["679593333241"]

  filter {
    name   = "name"
    values = ["openSUSE-Leap-16-0-v*hvm-ssd-x86_64-*"]
  }

  filter {
    name   = "virtualization-type"
    values = ["hvm"]
  }

  filter {
    name   = "state"
    values = ["available"]
  }
}

resource "aws_key_pair" "rancher" {
  count      = var.public_key_path != "" ? 1 : 0
  key_name   = "key-${var.combined_name}"
  public_key = file(var.public_key_path)
  tags       = var.tags
}

resource "aws_instance" "rancher" {
  ami                    = data.aws_ami.opensuse.id
  instance_type          = var.instance_type
  subnet_id              = var.subnet_id
  vpc_security_group_ids = [var.security_group_id]
  key_name               = var.public_key_path != "" ? aws_key_pair.rancher[0].key_name : var.key_name

  root_block_device {
    volume_size           = var.root_volume_size_gb
    volume_type           = "gp3"
    delete_on_termination = true
    tags                  = merge(var.tags, { Name = "ebs-${var.combined_name}" })
  }

  metadata_options {
    http_endpoint               = "enabled"
    http_tokens                 = "required"
    http_put_response_hop_limit = 2
  }

  tags = merge(var.tags, { Name = "ec2-${var.combined_name}" })

  lifecycle {
    ignore_changes = [ami]
  }
}

resource "aws_eip" "rancher" {
  instance = aws_instance.rancher.id
  domain   = "vpc"
  tags     = merge(var.tags, { Name = "eip-${var.combined_name}" })
}
